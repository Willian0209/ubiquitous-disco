// Inicia o git (repositório) no seu projeto
git init

// verificar alterações de pastas e arquivos no projeto
git status

// adiciona todos os arquivos e pastas modificados, ao stage area
git add .

// cria e descreve um ponto na história
git commit -m "message here"

// histórico de comimits do projeto
git log

// puxa modificações do repositório remoto para o repositório local (sempre utilizar esse aqui antes do git push)
git pull

// envia as modificações locais para o repositório remoto (para utilizar esse, deve sempre descrever um ponto na história com o git commit)
git push

// serve para baixar o repositório a partir de outro dispositivo (por exemplo se estiver em viagem com outro aparelho)
git clone (para utilizar, ir no repositório e clicar em code > selecionar HTTPS e copiar o link)
por exemplo: git clone https://github.com/Willian0209/ubiquitous-disco.git
